## Placeholder files

<!-- ===META=== one short paragraph on how to read them, then one short paragraph on their lifecycle; no list -->
<!-- + extensionless placeholder files signal the default "task" of each folder -->
<!-- + the placeholders are extensionless files, not .txt files; the filename is the signal and any body text elaborates -->

Each top-level folder holds a single extensionless file — `Archive`, `HITL`, `Orchestration`, `Runs`, `Ideas`, `Visuals`, `Tests`. These are not text files with a dropped extension; the filename itself is the payload, naming the default task the folder is there to serve. Where a body exists it only expands the name into a phrase, and an empty body means the name was judged sufficient on its own.

Treat a placeholder as a marker of intent rather than of content: it says what the folder is for before anything has been built there. Once a folder holds real contents its placeholder has done its job and should be removed, so the remaining placeholders always show what is still outstanding.

const app = document.getElementById('app');
const state = [];

function topLevel(p){
  const s = (p||'').trim().replace(/^\/+/,'');
  if(!s) return '';
  return s.split('/')[0] + '/';
}

function render(){
  app.innerHTML = '';
  DATA.forEach((grp, gi) => {
    const sec = document.createElement('section');
    const n = grp.items.reduce((a,b)=>a+b.n,0);
    sec.innerHTML = '<h2>' + grp.g + ' <span class="cnt">' + grp.items.length + ' entries · ' + n + ' files</span></h2>' +
                    (grp.blurb ? '<p class="sub">' + grp.blurb + '</p>' : '');
    const rows = document.createElement('div');
    rows.className = 'rows';
    grp.items.forEach((it, ii) => {
      const key = gi + ':' + ii;
      if(!(key in state)) state[key] = { on: !grp.off, dest: it.dest };
      const row = document.createElement('div');
      row.className = 'row' + (state[key].on ? '' : ' off');
      const tags = (it.tags||[]).map(t => '<span class="tag ' + t + '">' + t + '</span>').join('');
      row.innerHTML =
        '<div class="chk"><input type="checkbox" ' + (state[key].on ? 'checked' : '') + '></div>' +
        '<div class="src"><div class="p">' + it.src + '</div><div class="m">' + tags + it.note + '</div></div>' +
        '<div class="dst"><input type="text" value="' + state[key].dest.replace(/"/g,'&quot;') + '" placeholder="destination path"><div class="tl"></div></div>';
      const cb = row.querySelector('input[type=checkbox]');
      const tx = row.querySelector('input[type=text]');
      const tl = row.querySelector('.tl');
      const paint = () => {
        const t = topLevel(tx.value);
        tl.innerHTML = t ? 'groups under <b>' + t + '</b>' : '<span>no destination — will list as not moving</span>';
      };
      cb.addEventListener('change', () => { state[key].on = cb.checked; row.classList.toggle('off', !cb.checked); kpis(); });
      tx.addEventListener('input', () => { state[key].dest = tx.value; paint(); kpis(); });
      paint();
      rows.appendChild(row);
    });
    sec.appendChild(rows);
    app.appendChild(sec);
  });
  kpis();
}

function kpis(){
  let moving = 0, files = 0, dests = new Set(), skipped = 0;
  DATA.forEach((grp, gi) => grp.items.forEach((it, ii) => {
    const st = state[gi + ':' + ii];
    if(st && st.on && st.dest.trim()){ moving++; files += it.n; dests.add(topLevel(st.dest)); }
    else skipped++;
  }));
  document.getElementById('kpis').innerHTML =
    '<div class="kpi"><b>' + moving + '</b><span>entries moving</span></div>' +
    '<div class="kpi"><b>' + files + '</b><span>files moving</span></div>' +
    '<div class="kpi"><b>' + dests.size + '</b><span>destination folders</span></div>' +
    '<div class="kpi"><b>' + skipped + '</b><span>staying put</span></div>';
}

function buildText(){
  const byDest = new Map();
  const notMoving = [];
  DATA.forEach((grp, gi) => grp.items.forEach((it, ii) => {
    const st = state[gi + ':' + ii];
    if(!st) return;
    if(st.on && st.dest.trim()){
      const t = topLevel(st.dest);
      if(!byDest.has(t)) byDest.set(t, []);
      byDest.get(t).push('- `' + it.src + '` → `' + st.dest.trim() + '`  (' + it.n + ')');
    } else {
      notMoving.push('- `' + it.src + '`');
    }
  }));
  let out = '## FlightDeck transfer map — my choices\n';
  [...byDest.keys()].sort().forEach(t => {
    out += '\n### ' + t + '\n' + byDest.get(t).join('\n') + '\n';
  });
  if(notMoving.length) out += '\n### Not moving\n' + notMoving.join('\n') + '\n';
  return out;
}

function toast(){
  const t = document.getElementById('toast');
  t.classList.add('on');
  setTimeout(() => t.classList.remove('on'), 1600);
}

document.getElementById('copy').addEventListener('click', async () => {
  const text = buildText();
  try {
    await navigator.clipboard.writeText(text);
    toast();
  } catch(e) {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.style.position = 'fixed';
    ta.style.opacity = '0';
    document.body.appendChild(ta);
    ta.select();
    try { document.execCommand('copy'); toast(); }
    catch(e2){ alert('Copy failed — select the text manually:\n\n' + text); }
    document.body.removeChild(ta);
  }
});

  render();
});

## Vocabulary

- **FlightDeck** — the tool itself. Written `flightdeck/` when the scaffold directory it installs is meant.
- **keep file** — a `<task>.keep` file naming and describing the default task of the folder holding it.
- **deck** — a single installed `flightdeck/` scaffold within a project. A project may hold more than one.
- **launch** — one self-contained orchestrated build run, with its own specs and checks.
- **manuals vs library** — `manuals/` is what the crew reads to do its job; `library/` is information and records that persist over time, written for humans.

## Keep files

Each folder that needs describing holds a `<task>.keep` file — `archive.keep`, `orchestration.keep`, `runners.keep`, `checks.keep`, `agents.keep`, `workspace.keep`, `runs.keep`, `documentation.keep`, `ideas.keep`, `visuals.keep`, `tests.keep`. The filename names the default task the folder serves, and the body describes it in a line or two.

Unlike a conventional `.keep`, these are not disposable markers for an empty directory. They persist alongside real contents, so `flightcrew/checks/checks.keep` sits next to the scripts it describes. Read them first when working in an unfamiliar folder: several carry rules that are not inferable from the tree, such as `testbench/tests.keep` reserving `runs/` for local test results and requiring merge protection on the folder.
